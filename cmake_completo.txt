   144	if (NOT MSVC)
   145	    # Calls between this library's own translation units must land in this
   146	    # library. Without this, an internal call to an exported gl* symbol goes
   147	    # through a preemptible PLT slot, and in an Android app process the system
   148	    # libGLESv2 -- always in the global symbol scope -- can win the lookup: the
   149	    # DSA wrappers and the benchmark would then be calling the real driver where
   150	    # they meant the frontend, silently bypassing all state tracking. The
   151	    # symbols stay exported for dlsym and eglGetProcAddress; only who internal
   152	    # calls bind to changes. On the target, not in the AppleClang-guarded block
   153	    # above -- that block's condition has never been true (dangling NOT MATCHES),
   154	    # which is also why --gc-sections never made it into the link.
   155	    target_link_options(${CMAKE_PROJECT_NAME} PRIVATE -Wl,-Bsymbolic-functions)
   156	endif()
